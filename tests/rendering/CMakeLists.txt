cmake_minimum_required(VERSION 3.21)
project(HoloNightIconChecks LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Svg)
foreach(component Core Gui Svg)
  if(NOT TARGET Qt6::${component})
    message(FATAL_ERROR "Qt >=6.11 is missing required target Qt6::${component}")
  endif()
endforeach()
set(HOLONIGHT_RENDERER_FIXTURE "" CACHE PATH "Explicitly prepared approved IconRenderer fixture")
if(NOT HOLONIGHT_RENDERER_FIXTURE)
  message(FATAL_ERROR "Run task deps and supply HOLONIGHT_RENDERER_FIXTURE from build/deps/renderer")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/renderer-fixture.json" fixture_identity)
foreach(name iconrenderer.cpp iconrenderer.h)
  string(JSON expected GET "${fixture_identity}" files "src/icons/${name}")
  set(source "${HOLONIGHT_RENDERER_FIXTURE}/src/icons/${name}")
  if(NOT EXISTS "${source}")
    message(FATAL_ERROR "Missing prepared renderer file: ${name}; run task deps")
  endif()
  file(SHA256 "${source}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Incompatible renderer fixture: ${name}; run task deps")
  endif()
endforeach()
add_executable(render-check render_check.cpp "${HOLONIGHT_RENDERER_FIXTURE}/src/icons/iconrenderer.cpp")
target_include_directories(render-check PRIVATE "${HOLONIGHT_RENDERER_FIXTURE}/src/icons")
target_link_libraries(render-check PRIVATE Qt6::Core Qt6::Gui Qt6::Svg)
enable_testing()
add_test(NAME palette-rendering COMMAND render-check "${CMAKE_CURRENT_SOURCE_DIR}/../../build")
set_tests_properties(palette-rendering PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=")
