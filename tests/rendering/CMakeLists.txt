cmake_minimum_required(VERSION 3.21)
project(HoloNightIconChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Svg)
set(HOLONIGHT_QT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../holonight-qt" CACHE PATH "holonight-qt checkout containing the actual IconRenderer")
if(NOT EXISTS "${HOLONIGHT_QT_SOURCE}/src/icons/iconrenderer.cpp")
  message(FATAL_ERROR "Set HOLONIGHT_QT_SOURCE to a holonight-qt checkout. No KDE libraries are needed.")
endif()
add_executable(render-check render_check.cpp "${HOLONIGHT_QT_SOURCE}/src/icons/iconrenderer.cpp")
target_include_directories(render-check PRIVATE "${HOLONIGHT_QT_SOURCE}/src/icons")
target_link_libraries(render-check PRIVATE Qt6::Core Qt6::Gui Qt6::Svg)
enable_testing()
add_test(NAME palette-rendering COMMAND render-check "${CMAKE_CURRENT_SOURCE_DIR}/../../build")
set_tests_properties(palette-rendering PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=")
